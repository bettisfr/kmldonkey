cmake_minimum_required(VERSION 3.22)
project(KMLDONKEY VERSION 2.1.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network Xml Core5Compat)
find_package(KF6 6.3 REQUIRED COMPONENTS Config CoreAddons DBusAddons Codecs I18n XmlGui WidgetsAddons Completion KIO Notifications IconThemes)
include_directories(${CMAKE_SOURCE_DIR}/compat)
add_subdirectory(libkmldonkey)
add_subdirectory(kmldonkey)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_subdirectory(tests)
endif()

include(cmake/Packaging.cmake)
